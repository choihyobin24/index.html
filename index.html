let isSpinning = false;
function spinSnackRoulette() {
if (isSpinning) return;
isSpinning = true;
const display = document.getElementById('snack-display');
const side = document.getElementById('snack-side');
const btn = document.getElementById('btn-spin');
btn.disabled = true;
side.innerText = "";
let counter = 0;
const spinInterval = setInterval(() => {
const randomItem = snacks[Math.floor(Math.random() * snacks.length)];
display.innerText = randomItem.name;
counter++;
if (counter > 18) {
clearInterval(spinInterval);
const finalChoice = snacks[Math.floor(Math.random() * snacks.length)];
display.innerText = finalChoice.name;
side.innerText = `💡 꿀조합: ~${finalChoice.combo}`;
isSpinning = false;
btn.disabled = false;
}
}, 90);
}
// ================= CHAPTER 4: N-SPLIT LOGIC =================
function calculateSplit() {
const total = parseInt(document.getElementById('split-total').value) || 0;
const people = parseInt(document.getElementById('split-people').value) || 1;
const payer = document.getElementById('split-payer').value || '총무';
const account = document.getElementById('split-account').value;
const perPerson = Math.floor(total / people);
const remainder = total % people;
document.getElementById('split-per-person').innerText = `${perPerson.toLocaleString()}원`;
document.getElementById('split-rem').innerText = remainder > 0 ? `(잔액 ${remainder}원은 선결제자 몫!)` : '';
let msg = `[🍔 맛집/N빵 정산]\n`;
msg += `• 총 금액: ${total.toLocaleString()}원 (${people}명)\n`;
msg += `• 1인당 보내줄 금액: ${perPerson.toLocaleString()}원\n`;
msg += `• 입금할 닉네임: ${payer}\n`;
if (account) msg += `• 송금정보: ${account}\n`;
msg += `맛있게 잘 먹었습니다! 👍`;
document.getElementById('split-text').value = msg;
}

}
